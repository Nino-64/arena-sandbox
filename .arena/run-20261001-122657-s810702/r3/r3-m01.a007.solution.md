# Pulse Orbit: a one-button neon arcade game

You get one file, `index.html`, with no dependencies, no build step and no network requests. Save it and open it in any modern browser (desktop or mobile, portrait or landscape) and it plays straight away.

**How it plays:** your comet orbits a pulsing core on one of two rings. Tap, click or press Space to switch rings. Gold shards build your **chain**, and every 5 in the chain raises the score multiplier (up to ×9). Red spikes flash a dashed warning first and can't hurt you during it. Then they go live. If you hit one, the run ends. If you were really riding a spike's ring less than 0.35 s before you pass it, and you got off in time, you score a **CLUTCH** bonus. This works on every pass, so a spike that is still alive when you lap back round to it can be clutched again (a quick double-tap that never leaves your own ring scores nothing). Every 10th shard in a chain fires a **PULSE**: a shockwave clears every spike, time slows for a moment and your orbit reverses. If you let a shard expire (it blinks first), your chain breaks. Speed, spawn rate and music tempo keep rising. Shards you collect are kept between runs and buy skins.

Controls: **Tap / Click / Space / ↑ / W** switch orbit · **P / Esc** pause · **R** restart · **M** mute. The game pauses itself when the tab loses focus. After a crash, tap or press Space/R again from 0.3 s onward to go straight into the next run; the run is saved first and the new run opens with a "NEW BEST · score" / "+N ◆ saved" banner so you still see what you earned. **Every way a run can end is saved:** crash, R, pause → Restart, pause → Menu, and closing the tab.

  stop() { this.on = false; this.beats.length = 0; this.sleep(); },
  schedule() {
    if (!this.on || !A.ctx || A.ctx.state !== 'running') return;
    const c = A.ctx;
    if (this.next < c.currentTime - 0.05) this.next = c.currentTime + 0.02;
    while (this.next < c.currentTime + 0.12) {
      this.play(this.step, this.next);
      this.next += 60 / this.bpm / 4;
      this.step = (this.step + 1) % 64;
    }
  },
  play(s, t) {
    const root = BASS[Math.floor(s / 16)];
    const b = A.mus;
    if (s % 4 === 0) {
      A.tone(150, 0.2, { type: 'sine', vol: 0.75, slide: 42, when: t, bus: b });
      this.beats.push(t);
    }
    if (s % 2 === 1) A.noise(0.04, { ft: 'highpass', ff: 7500, vol: 0.12, when: t, bus: b });
    if (this.layer >= 1 && s % 8 === 4) A.noise(0.14, { ft: 'bandpass', ff: 1800, q: 1.2, vol: 0.22, when: t, bus: b });
    if (s % 2 === 0) {
      const oct = (s % 8 === 6) ? 12 : 0;
      A.tone(noteHz(root + oct, 55), 0.17, { type: 'sawtooth', vol: 0.16, lp: 520 + this.layer * 260, when: t, bus: b });
    }
    if (this.layer >= 2) {
      const chord = [0, 3, 7, 10, 12, 7];
      A.tone(noteHz(root + chord[s % 6], 220), 0.11, { type: 'square', vol: 0.035, lp: 3200, when: t, bus: b });
    }
  }
};

function burst(x, y, n, col, spd = 260, size = 3, life = 0.7) {
  for (let i = 0; i < n; i++) {
    const a = Math.random() * TAU, s = spd * U * (0.2 + Math.random() * 0.8), l = life * (0.5 + Math.random() * 0.5);
    const p = { x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: l, max: l, size: size * U * (0.6 + Math.random() * 0.8), col };
    // At the 800 cap, overwrite the particle closest to dying (sampled, O(1)) instead of re-indexing the array.
    if (parts.length < MAX_PARTS) parts.push(p);
    else { let j = (Math.random() * MAX_PARTS) | 0; const k = (Math.random() * MAX_PARTS) | 0; if (parts[k].life < parts[j].life) j = k; parts[j] = p; }
  }
}

function swap() {

const hazardGap = () => Math.max(0.45, G.omega * 0.3);
function spawnHazard() {
  if (G.objs.filter(o => o.k === 'h').length >= 14) return;
  const a = G.p.a + G.dir * clamp(G.omega * rand(1.0, 1.7), 1.15, 4.2);
  let ring = Math.random() < 0.5 ? 0 : 1;
  const gap = hazardGap(); // always leave a passable gap on the other orbit
  if (G.objs.some(o => o.k === 'h' && o.ring !== ring && Math.abs(wrap(o.a - a)) < gap)) ring = 1 - ring;
  if (G.objs.some(o => o.k === 'h' && o.ring !== ring && Math.abs(wrap(o.a - a)) < gap)) return;
  if (G.objs.some(o => o.ring === ring && Math.abs(wrap(o.a - a)) < 0.3)) return;
  const moving = G.t > 18 && Math.random() < Math.min(0.4, (G.t - 18) * 0.012);
  G.objs.push({ k: 'h', ring, a, va: moving ? rand(0.2, 0.45) * (Math.random() < 0.5 ? 1 : -1) : 0,
    age: 0, life: rand(6, 8), spin: rand(-3, 3), prevD: null });
}
function spawnGem() {
  if (G.objs.filter(o => o.k === 'g').length >= 3) return;
  const a = G.p.a + G.dir * clamp(G.omega * rand(0.9, 1.6), 1.0, 4.0);
  const ring = Math.random() < 0.5 ? 0 : 1;
  if (G.objs.some(o => Math.abs(wrap(o.a - a)) < (o.k === 'g' ? 0.6 : o.ring === ring ? 0.35 : 0))) return;
  G.objs.push({ k: 'g', ring, a, va: 0, age: 0, life: 4.6 });
}

function pulse() {
  G.pulses++;
  G.shock = { r: 0 };
  for (const o of G.objs) if (o.k === 'h') {
    const r = o.ring ? R1 : R0;
    burst(CX + Math.cos(o.a) * r, CY + Math.sin(o.a) * r, 18, '#ff3d7f', 300, 3, 0.8);
  }
  G.objs = G.objs.filter(o => o.k !== 'h');
  G.dir *= -1; G.slow = 0.4; G.hzT = 1.1;
  G.score += 50 * mult();
  shake(0.55); flash = 0.5; flashCol = skinCol();
  popText(CX, CY - 70 * U, 'PULSE · REVERSE', skinCol(), 26);
  SFX.pulse();
}

}

function update(dt) {
  G.t += dt;
  G.omega = Math.min(3.3, 1.5 + G.t * 0.02);
  Music.bpm = 112 + (G.omega - 1.5) * 14;
  Music.layer = G.chain >= 5 ? 2 : G.t > 12 ? 1 : 0;
  const P = G.p;
  P.a += G.omega * G.dir * dt;
  P.blend += (P.ring - P.blend) * Math.min(1, dt * 22);
  if (P.blend < 0.2) P.onT[0] = G.t;      // comet is genuinely sitting on the inner orbit
  if (P.blend > 0.8) P.onT[1] = G.t;      // ... or on the outer orbit
  const pr = lerp(R0, R1, P.blend);
  P.x = CX + Math.cos(P.a) * pr; P.y = CY + Math.sin(P.a) * pr;

  G.hzT -= dt; if (G.hzT <= 0) { spawnHazard(); G.hzT = Math.max(0.42, 1.35 - G.t * 0.012) * rand(0.75, 1.25); }
  G.gmT -= dt; if (G.gmT <= 0) { spawnGem(); G.gmT = rand(0.9, 1.4); }

  let doPulse = false;
  const gap = hazardGap();
  for (let i = G.objs.length - 1; i >= 0; i--) {
    const o = G.objs[i];
    o.age += dt;
    if (o.va) {
      // A drifting spike bounces off any spike on the other orbit before the two can
      // overlap in angle, so a passable gap on one ring is kept for its whole life.
      const na = o.a + o.va * dt;
      if (G.objs.some(q => q.k === 'h' && q.ring !== o.ring && Math.abs(wrap(na - q.a)) < gap && Math.abs(wrap(na - q.a)) < Math.abs(wrap(o.a - q.a)))) o.va = -o.va;
      else o.a = na;
    }
    const r = o.ring ? R1 : R0, ox = CX + Math.cos(o.a) * r, oy = CY + Math.sin(o.a) * r;
    const dist = Math.hypot(ox - P.x, oy - P.y);
    if (o.k === 'h') {
      if (o.age > o.life + 0.35) { G.objs.splice(i, 1); continue; }
      const active = o.age > TELE && o.age < o.life;
      if (active && dist < (PR + HR * 0.75) * U) { die('SPIKED'); return; }
      const d = wrap(o.a - P.a) * G.dir;
      // Fires once per pass: d falls through 0 exactly once per lap (spike drift 0.45 rad/s max < orbit speed 1.5 rad/s min),
      // so a spike that is still alive when you lap it again can award another CLUTCH, same as it can still kill you.
      if (o.prevD !== null && o.prevD > 0 && d <= 0 && Math.abs(d) < 1) {
        // CLUTCH = you really were on this spike's orbit less than 0.35 s ago and have now left it.
        // A double-tap never moves the comet onto the other orbit, so it cannot farm this.
        if (active && o.ring !== P.ring && Math.abs(P.blend - o.ring) > 0.6 && G.t - P.onT[o.ring] < 0.35) {
          G.clutches++;
          const pts = 25 * mult(); G.score += pts;
          popText(P.x, P.y - 26 * U, 'CLUTCH +' + pts, '#ff3d7f', 20);
          burst(ox, oy, 14, '#ff3d7f', 220, 2.5, 0.5);
          shake(0.18); SFX.clutch();
        }
      }
      o.prevD = d;
    } else {
      if (o.age > 0.2 && dist < (PR + GR + 8) * U) {
        G.objs.splice(i, 1);
        G.chain++; G.gems++; G.bestChain = Math.max(G.bestChain, G.chain);
        const pts = 10 * mult(); G.score += pts;
        burst(ox, oy, 22, '#ffd166', 240, 3, 0.6);
        burst(ox, oy, 8, skinCol(), 160, 2, 0.4);
        popText(ox, oy - 22 * U, '+' + pts, '#ffd166', 18);
        shake(0.08); SFX.gem(G.chain);
        if (G.chain % 10 === 0) doPulse = true;
        continue;
      }
      if (o.age > o.life) {
        G.objs.splice(i, 1);
        if (G.chain > 0) {
          popText(ox, oy - 18 * U, 'CHAIN BROKEN', '#8d88a6', 15);
          G.chain = 0; SFX.breakChain();
        }
      }
    }
  }
  if (doPulse) pulse();
  if (G.shock) { G.shock.r += dt * 900 * U; if (G.shock.r > Math.hypot(W, H)) G.shock = null; }
}

/* Saves whatever this run has earned since the last bank. Safe to call any number of times:
   only the difference is added, so a checkpoint (tab hidden) followed by the crash never double-counts.
   final = the run is over (crash, restart, menu, page closed): it then also counts as a play. */
function bank(final) {
  if (!G) return;
  const b = G.bk;
  const dScore = G.score - b.score, dGems = G.gems - b.gems, dPulses = G.pulses - b.pulses, dClutch = G.clutches - b.clutches;
  const before = levelInfo(S.xp).lvl;
  S.xp += dScore;
  const lv = (levelInfo(S.xp).lvl - before) * 20;
  const earned = dGems + dPulses * 5 + dClutch * 2 + lv;
  S.shards += earned; S.gems += dGems;
  S.bestChain = Math.max(S.bestChain, G.bestChain);
  if (G.score > S.best) S.best = G.score;
  G.bk = { score: G.score, gems: G.gems, pulses: G.pulses, clutches: G.clutches };
  G.earned += earned; G.lvBonus += lv;
  if (final && !G.counted) { G.counted = true; S.plays++; }
  save();
}
const runLive = () => G && !G.counted && (state === 'play' || state === 'paused' || state === 'dying');

function endRun() {
  state = 'over';
  bank(true);
  G.ended = true;
  const before = levelInfo(G.startXp).lvl, li = levelInfo(S.xp);
  const earned = G.earned, lvBonus = G.lvBonus;
  const isBest = G.score > G.startBest;

  $('#hud').hidden = true;
  $('#oCause').textContent = (G.cause || 'ORBIT BROKEN') + ' · ' + G.t.toFixed(1) + 's';
  $('#oBadge').hidden = !isBest;
  $('#oBest').textContent = S.best;
  $('#oChain').textContent = G.bestChain;
  $('#oClutch').textContent = G.clutches;
  $('#oLevel').textContent = 'LEVEL ' + li.lvl + (li.lvl > before ? '  ▲' : '');
  $('#oXp').textContent = li.cur + ' / ' + li.need + ' XP';
  const bar = $('#oBar'); bar.style.width = '0%';
  requestAnimationFrame(() => requestAnimationFrame(() => { bar.style.width = (li.cur / li.need * 100).toFixed(1) + '%'; }));
  $('#oUnlock').innerHTML = unlockLine(earned, lvBonus);
  countUp($('#oScore'), G.score);
  showScreen('#over');
  overAt = performance.now();
  if (isBest) SFX.fanfare();
}
function unlockLine(earned, lvBonus) {
  let s = '<span class="shard">+' + earned + ' ◆</span>' + (lvBonus ? ' (incl. ' + lvBonus + ' level-up bonus)' : '');
  const next = SKINS.filter(k => !S.owned.includes(k.id)).sort((a, b) => a.cost - b.cost)[0];
  if (!next) s += ' · Every skin unlocked.';
  else if (S.shards >= next.cost) s += ' · <strong>' + next.name + '</strong> is ready to unlock!';
  else s += ' · <strong>' + (next.cost - S.shards) + '</strong> more to unlock <strong>' + next.name + '</strong>';
  return s;
}
function countUp(el, to) {
  const t0 = performance.now(), dur = Math.min(900, 250 + to * 2);
  const step = now => {
    const k = Math.min(1, (now - t0) / dur);
    el.textContent = Math.round(to * (1 - Math.pow(1 - k, 3)));
    if (k < 1 && state === 'over') requestAnimationFrame(step); else el.textContent = to;
  };
  requestAnimationFrame(step);
}

## Fairness rules built in
- Spikes always spawn at least about 1 s of travel ahead of you. They are harmless during their 0.65 s warning.
- A new spike is never placed next to a spike on the other ring unless there is a gap of at least `hazardGap()` = `max(0.45 rad, 0.3 s of travel)`.
- The gap also holds for the spike's whole life, not only when it spawns. A drifting spike (after 18 s) reverses before it can come closer than `hazardGap()` to a spike on the other ring, so the two rings are never blocked at the same angle (checked with a 20,000-pair random simulation: closest cross-ring separation 0.450 rad).
- Movement and collision are sub-stepped (`frame()`). No single step moves the comet further than its own radius, so a live spike can't be skipped at a low frame rate (dt clamp 50 ms) or on a large screen. Worst-case step at 20 fps and top speed: 5.5 px against an 11.4 px hit radius at 375×667, and 12.9 px against 26.6 px at 3840×2160.
- CLUTCH can't be farmed: it requires that the comet was physically on the spike's orbit (`P.onT[ring]`, set only while the ring blend is within 0.2 of that orbit) less than 0.35 s before passing, and is now clear of it (blend more than 0.6 away). Tested: a 20 ms double-tap past a live spike on the other ring gives 0 clutches; really riding the spike's ring and switching off 0.28 s before it gives 1 clutch and no death.
- CLUTCH is scored per pass, not once per spike: the reward check runs on every crossing, exactly like the collision check. A crossing is detected when the signed angle to the spike falls through 0, which happens once per lap because a spike drifts at most 0.45 rad/s and the comet never orbits slower than 1.5 rad/s. Tested: riding and leaving the same long-lived spike on three consecutive laps scored 3 clutches (at 0.58 s, 4.63 s and 8.48 s).
- The collision radius is smaller than the spike you see, and the shard pickup radius is larger than the shard, so near misses feel generous.

## Assumptions
- "Unlockables" means cosmetic skins bought with shards earned in play, plus account levels that pay out shards. There is no gameplay pay-to-win.
- Browsers only allow audio after a user gesture, so sound starts on your first tap or key press. That is standard Web Audio behaviour, not a bug.
- If `localStorage` is blocked (some private modes), the game still runs. Progress then lasts only for that session.
- Tested only in headless Chromium (with touch/mobile emulation for the phone viewports). Safari and Firefox support every API used (Canvas 2D, Web Audio with the `webkitAudioContext` fallback, Pointer Events, `pagehide`), but I did not run it in them. Mobile browsers may kill a backgrounded tab without firing `pagehide`; the checkpoint taken on `visibilitychange` covers that case up to the moment the tab was hidden.
